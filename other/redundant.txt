static void arrange_vertices_CCW(p3 *VA) {
    p3 arranged_VA[3];
    size_t max_index = get_highest_point(VA);
    arranged_VA[0] = VA[max_index];
    double max_x = NV;

    for (int i = 0; i < 3; i++) {
        if (i != max_index && (max_x == NV || VA[i].x <= max_x)) {
            max_x = VA[i].x;
            arranged_VA[1] = VA[i];
        } else if (i != max_index && (VA[i].x >= max_x)) {
            arranged_VA[2] = VA[i];
        }
    }
}

static size_t get_highest_point(p3 *VA) {
    p3 max = VA[0];
    size_t index = 0;
    for (int i = 1; i < sizeof(VA) / sizeof(VA[0]); i++) {
        if (VA[i].y < max.y && VA[i].y != NV) {
            max = VA[i];
            index = i;
        }
    }
    return index;
}

static size_t get_lowest_point(p3 *VA) {
    p3 min = VA[0];
    size_t index = 0;
    for (int i = 1; i < sizeof(VA) / sizeof(VA[0]); i++) {
        if (VA[i].y > min.y && VA[i].y != NV) {
            min = VA[i];
            index = i;
        }
    }
    return index;
}
